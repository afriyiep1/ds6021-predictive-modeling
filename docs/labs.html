# Questions 3a--3c: Pairs bootstrap for Model 2.
rng = np.random.default_rng(6021)
B = 1000
y_array = y.to_numpy()
coefficient_names = ["Intercept"] + list(feature_names)
bootstrap_estimates = np.empty(
    (B, len(coefficient_names))
)
for b in range(B):
    # Sample complete observations with replacement.
    indices = rng.integers(0, n, size=n)
    bootstrap_model = LinearRegression()
    bootstrap_model.fit(
        X_encoded[indices],
        y_array[indices]
    )
    bootstrap_estimates[b] = np.r_[
        bootstrap_model.intercept_,
        bootstrap_model.coef_
    ]
bootstrap_coefs = pd.DataFrame(
    bootstrap_estimates,
    columns=coefficient_names
)
# Questions 3a and 3b: Standard errors and percentile intervals.
bootstrap_summary = pd.DataFrame({
    "Estimate": coefs2,
    "Bootstrap mean": bootstrap_coefs.mean(),
    "Bootstrap SE": bootstrap_coefs.std(ddof=1),
    "CI lower": bootstrap_coefs.quantile(0.025),
    "CI upper": bootstrap_coefs.quantile(0.975)

def summarize_fit(name, observed, fitted, p):
    n_obs = len(observed)
    r2 = r2_score(observed, fitted)
    adjusted_r2 = (
        1 - (1 - r2) * (n_obs - 1) / (n_obs - p - 1)
    )
    return {
        "Model": name,
        "n": n_obs,
        "p": p,
        "R2": r2,
        "Adjusted R2": adjusted_r2
    }

    summarize_fit(
        "Model 1: numerical",
        y_array,
        fitted1,
        X_numeric.shape[1]
    ),
    summarize_fit(
        "Model 2: numerical + categorical",
        y_array,
        fitted2,
        X_encoded.shape[1]
    )
]).set_index("Model")

Stage Completed activity
1. Form a team Select a team of 3–4 members.
2. Define the problem Select a problem space and identify likely datasets.
3. Understand the data Explore, clean, and handle missing data where appropriate.
4. Establish a baseline Fit and evaluate at least one classification or regression model.
5. Compare and build Try at least two modeling techniques and begin an end-to-end app.
6. Extend and communicate Investigate dimensionality reduction or unsupervised learning as appropriate, then prepare the presentation.
